CLASS Transaction {
    ATTRIBUTES:
        transactionId
        merchant
        month
        amount

    CONSTRUCTOR(transactionId, merchant, month, amount) {
        SET this.transactionId = transactionId
        SET this.merchant = merchant
        SET this.month = month
        SET this.amount = amount
    }
}

MAIN METHOD {
    CREATE transactions LIST OF Transaction
    ADD new Transaction("T1", "Amazon", "Jan", 2000) INTO transactions
    ADD new Transaction("T2", "Flipkart", "Jan", 1500) INTO transactions
    ADD new Transaction("T3", "Amazon", "Feb", 3000) INTO transactions
    ADD new Transaction("T4", "Flipkart", "Feb", 1200) INTO transactions

    REPORT = GROUP transactions BY merchant AND month
             CALCULATE totalAmount = SUM(amount)

    FOR EACH entry IN REPORT {
        PRINT "Merchant: " + entry.merchant +
              ", Month: " + entry.month +
              ", Total: " + entry.totalAmount
    }
}

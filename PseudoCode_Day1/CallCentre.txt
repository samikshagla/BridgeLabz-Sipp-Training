Initialize QUEUE normalQueue
Initialize QUEUE emergencyQueue

Function addCustomer(name, priority):
    IF priority == "emergency"
        ENQUEUE name into emergencyQueue
    ELSE
        ENQUEUE name into normalQueue

Function serveCustomer():
    IF emergencyQueue not empty
        customer = DEQUEUE from emergencyQueue
    ELSE IF normalQueue not empty
        customer = DEQUEUE from normalQueue
    ELSE
        PRINT "No customers"
        RETURN
    PRINT "Serving: " + customer

Main:
    addCustomer("John", "normal")
    addCustomer("Alex", "emergency")
    serveCustomer()  
    serveCustomer()

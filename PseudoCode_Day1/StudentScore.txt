Initialize MAP studentScores

Add ("Alice", 85)
Add ("Bob", 92)
Add ("Charlie", 78)

Function findHighestScorer():
    IF studentScores is empty
        PRINT "No scores available"
        RETURN
    SET bestName = ""
    SET bestScore = -infinity
    FOR each (name, score) in studentScores
        IF score > bestScore
            bestScore = score
            bestName = name
    PRINT "Highest Scorer: " + bestName + " (" + bestScore + ")"

Main:
    CALL findHighestScorer()

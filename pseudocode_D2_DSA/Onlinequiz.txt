CLASS QuizTimer {
    ATTRIBUTES:
        timeSlots[]   // circular array
        size

    CONSTRUCTOR(slots[]) {
        SET timeSlots = slots
        SET size = LENGTH of slots
    }

    METHOD startQuiz() {
        FOR i = 0 TO size-1 {
            PRINT "Question " + (i+1) + " started."
            WAIT timeSlots[i] seconds
            IF timeSlots[i] < 5 THEN
                PRINT "Warning: Time almost up!"
            END IF
        }
    }
}

MAIN METHOD {
    CREATE timer OBJECT OF QuizTimer([30, 25, 20])
    CALL timer.startQuiz()
}
